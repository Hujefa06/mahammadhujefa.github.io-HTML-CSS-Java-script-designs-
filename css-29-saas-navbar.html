/* Glassmorphism Frosted Header */

  top: 0;
  background: rgba(255, 255, 255, 0.85);

/* Hover-triggered Dropdown Menu */

  display: block;
  animation: slideDown 0.2s cubic-bezier(0.4, 0, 0.2, 1);
