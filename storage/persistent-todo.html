Persistent Task List (LocalStorage State)

A complete interactive Todo Application where newly created tasks, completed checkbox states, and deletions are synchronized permanently in localStorage, surviving page refreshes.

// 1. Load tasks from storage or initialize empty array
let tasks = JSON.parse(localStorage.getItem("todos_list") || "[]");

function saveAndRender() {
  // 2. Persist state as JSON string
  localStorage.setItem("todos_list", JSON.stringify(tasks));
  renderTasks();
}

function addTask(text) {
  tasks.push({ id: Date.now(), text, completed: false });
  saveAndRender();
}

function toggleTask(id) {
  const item = tasks.find(t => t.id === id);
  if (item) item.completed = !item.completed;
  saveAndRender();
}

function deleteTask(id) {
  tasks = tasks.filter(t => t.id !== id);
  saveAndRender();
}
