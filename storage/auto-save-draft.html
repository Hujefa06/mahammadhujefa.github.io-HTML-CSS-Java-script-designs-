Form Auto-Save Draft with Debounce

Automatically serialize and save form articles to localStorage 500ms after the user stops typing (debouncing). Guarantees zero data loss if the browser unexpectedly closes or crashes.

let debounceTimer = null;

function queueAutoSave() {
  saveStatus.textContent = "Saving changes...";
  saveDot.style.color = "#f59e0b"; // yellow while typing

  // Debounce: Wait 500ms after last keystroke before writing to disk
  clearTimeout(debounceTimer);
  debounceTimer = setTimeout(saveToStorage, 500);
}

function saveToStorage() {
  const draft = {
    title: titleInput.value,
    body: bodyInput.value,
    updatedAt: new Date().toLocaleTimeString()
  };

  localStorage.setItem("lab_assignment_draft", JSON.stringify(draft));
  saveStatus.textContent = `Draft saved at ${draft.updatedAt}`;
  saveDot.style.color = "#10b981"; // green
}
