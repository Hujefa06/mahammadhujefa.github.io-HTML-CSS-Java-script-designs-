Dynamic Form Field Repeater & Payload Serializer

Dynamically append and delete multi-column educational record rows (Degree, Institution, Year, GPA) before aggregating into a JSON array structure for API submission.

function addRow() {
  const container = document.getElementById("educationRows");
  const row = document.createElement("div");
  row.className = "field-row-item";
  row.innerHTML = `
    <input type="text" class="repeater-input row-degree" placeholder="Degree">
    <input type="text" class="repeater-input row-inst" placeholder="College">
    <input type="text" class="repeater-input row-year" placeholder="Year">
    <input type="text" class="repeater-input row-score" placeholder="GPA">
    <button type="button" class="btn-remove-row" onclick="this.parentElement.remove()">×</button>
  `;
  container.appendChild(row);
}

function serializeRows() {
  const rows = document.querySelectorAll(".field-row-item");
  const data = Array.from(rows).map(row => ({
    degree: row.querySelector(".row-degree").value,
    institution: row.querySelector(".row-inst").value,
    year: row.querySelector(".row-year").value,
    score: row.querySelector(".row-score").value
  }));
  return JSON.stringify(data, null, 2);
}
