function calculateAcademicGrade(marksArray) {
  const total = marksArray.reduce((acc, val) => acc + val, 0);
  const percentage = (total / (marksArray.length * 100)) * 100;
  const sgpa = (percentage / 10).toFixed(2);

  // Grade classification
  let grade = "F";
  if (percentage >= 90) grade = "O (Outstanding)";
  else if (percentage >= 80) grade = "A+ (Excellent)";
  else if (percentage >= 70) grade = "A (Very Good)";
  else if (percentage >= 60) grade = "B+ (Good)";
  else if (percentage >= 50) grade = "B (Above Average)";
  else if (percentage >= 40) grade = "C (Pass)";

  const isFailed = marksArray.some(m => m < 40);
  return { total, percentage, sgpa, grade, isFailed };
}
