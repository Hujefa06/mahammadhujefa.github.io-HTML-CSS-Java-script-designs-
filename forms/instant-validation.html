Instant On-The-Fly Input Validation

Provide immediate visual feedback as the user types using input and blur events. Dynamic border colors, status checkmarks/crosses, and contextual hints guide the user before submission.

function updateFieldVisuals(rowId, iconId, hintId, isValid, successMsg, failMsg) {
  const row = document.getElementById(rowId);
  const icon = document.getElementById(iconId);
  const hint = document.getElementById(hintId);

  if (isValid) {
    row.className = "field-row valid";
    icon.textContent = "✔";
    icon.style.color = "#10b981";
    hint.textContent = successMsg;
  } else {
    row.className = "field-row invalid";
    icon.textContent = "✖";
    icon.style.color = "#ef4444";
    hint.textContent = failMsg;
  }
}
