Password Entropy & Strength Meter

Calculate real-time password entropy scoring and reflect visual feedback via a multi-stage color-coded meter (Weak → Fair → Good → Strong).

Dynamic Scoring
Strength Level: Empty
💡 Scoring Criteria: Length (+25%), Lowercase (+15%), Uppercase (+20%), Digits (+20%), Special Characters (+20%). 
function evaluatePassword(pwd) {
  let score = 0;
  if (pwd.length >= 8) score += 25;
  if (pwd.length >= 12) score += 10;
  if (/[a-z]/.test(pwd)) score += 15;
  if (/[A-Z]/.test(pwd)) score += 20;
  if (/[0-9]/.test(pwd)) score += 15;
  if (/[^A-Za-z0-9]/.test(pwd)) score += 15;

  score = Math.min(100, score);

  let label = "Weak", color = "#ef4444";
  if (score > 80) { label = "Very Strong"; color = "#10b981"; }
  else if (score > 60) { label = "Strong"; color = "#22c55e"; }
  else if (score > 40) { label = "Moderate"; color = "#f59e0b"; }

  return { score, label, color };
}
