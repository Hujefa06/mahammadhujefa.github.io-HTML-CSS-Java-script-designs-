function runRegexTest(patternStr, flagsStr, testText) {
  try {
    const regex = new RegExp(patternStr, flagsStr);
    
    // Replace matches with custom HTML span tags for highlighting
    const highlighted = testText.replace(regex, (match) => {
      return `<mark class="match-highlight">${match}</mark>`;
    });

    const matches = testText.match(regex) || [];
    console.log(`Found ${matches.length} matches:`, matches);

    return { highlighted, matchCount: matches.length };
  } catch (err) {
    return { error: "Invalid Regular Expression syntax: " + err.message };
  }
}
