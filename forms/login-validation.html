function validateLogin(event) {
  event.preventDefault();
  let isValid = true;

  const email = document.getElementById("loginEmail").value.trim();
  const pwd = document.getElementById("loginPwd").value;

  // 1. Identifier Validation
  if (email.length < 3) {
    showError("loginEmail", "errLoginEmail");
    isValid = false;
  } else {
    clearError("loginEmail", "errLoginEmail");
  }

  // 2. Password Check
  if (pwd === "") {
    showError("loginPwd", "errLoginPwd");
    isValid = false;
  } else {
    clearError("loginPwd", "errLoginPwd");
  }

  if (isValid) {
    alert("Authentication credentials verified! Redirecting...");
  }
}
