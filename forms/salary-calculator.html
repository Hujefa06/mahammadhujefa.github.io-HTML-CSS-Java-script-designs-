function calculateSalary(basic, daPercent, hraPercent, bonus) {
  // 1. Earnings
  const da = (basic * daPercent) / 100;
  const hra = (basic * hraPercent) / 100;
  const gross = basic + da + hra + bonus;

  // 2. Deductions
  const pf = (basic * 12) / 100; // standard 12% PF
  const pt = 200; // Professional Tax
  const totalDeductions = pf + pt;

  // 3. Net In-Hand
  const netSalary = gross - totalDeductions;

  return { gross, netSalary, da, hra, pf, pt };
}
