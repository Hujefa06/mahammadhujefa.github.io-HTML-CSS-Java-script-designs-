Radio Buttons & Checkbox Group Validation

Validate choice groups in HTML: ensure at least one radio button in a name group is selected, enforce a minimum selection count for checkboxes (e.g. at least 2 technical skills), and require terms agreement.

Group Constraints
1. Degree Specialization Track (Mandatory 1):
Please select one specialization track.
2. Programming Languages (Select minimum 2):
You must choose at least 2 programming languages.
You must accept the terms to proceed.
function validateForm() {
  // 1. Radio check (returns null if none checked)
  const selectedRadio = document.querySelector('input[name="track"]:checked');
  if (!selectedRadio) {
    alert("Please select a track!");
    return false;
  }

  // 2. Multiple Checkbox check (query all matching)
  const checkedBoxes = document.querySelectorAll('input[name="skills"]:checked');
  if (checkedBoxes.length < 2) {
    alert("Please select at least 2 skills!");
    return false;
  }

  // 3. Single checkbox boolean property
  const termsAgreed = document.getElementById("termsCheck").checked;
  if (!termsAgreed) {
    alert("Please accept the terms!");
    return false;
  }

  return true;
}
