function calculateInvoice() {
  const q1 = parseInt(document.getElementById("qty1").value) || 0;
  const q2 = parseInt(document.getElementById("qty2").value) || 0;
  const q3 = parseInt(document.getElementById("qty3").value) || 0;

  const subtotal = (q1 * 2499) + (q2 * 899) + (q3 * 399);
  const discount = subtotal * 0.10; // 10% coupon discount
  const taxableAmount = subtotal - discount;
  const gst = taxableAmount * 0.18; // 18% GST
  const grandTotal = Math.round(taxableAmount + gst);

  return { subtotal, discount, gst, grandTotal };
}
