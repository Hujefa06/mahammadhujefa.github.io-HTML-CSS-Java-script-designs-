Student Registration Form Validation

Perform complete client-side form validation before dispatch: required check, regex email format, 10-digit mobile number, matching passwords, and inline field error feedback.

function validateRegistration(event) {
  event.preventDefault(); // Stop default form submit
  let isValid = true;

  // 1. Name Check (Min 3 chars)
  const name = document.getElementById("regName").value.trim();
  if (name.length < 3) {
    showError("regName", "errName");
    isValid = false;
  } else {
    showSuccess("regName", "errName");
  }

  // 2. Email Regex Check
  const email = document.getElementById("regEmail").value.trim();
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!emailRegex.test(email)) {
    showError("regEmail", "errEmail");
    isValid = false;
  } else {
    showSuccess("regEmail", "errEmail");
  }

  // 3. Mobile (10-digit check starting with 6-9)
  const phone = document.getElementById("regPhone").value.trim();
  if (!/^[6-9]\d{9}$/.test(phone)) {
    showError("regPhone", "errPhone");
    isValid = false;
  } else {
    showSuccess("regPhone", "errPhone");
  }

  // 4. Password Match Check
  const p1 = document.getElementById("regPassword").value;
  const p2 = document.getElementById("regConfirm").value;
  if (p1.length < 6 || p1 !== p2) {
    showError("regConfirm", "errConfirm");
    isValid = false;
  } else {
    showSuccess("regConfirm", "errConfirm");
  }

  if (isValid) {
    document.getElementById("successBox").style.display = "block";
  }
}
