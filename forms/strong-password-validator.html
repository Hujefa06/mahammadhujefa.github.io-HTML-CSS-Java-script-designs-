🛡️ Real-Time Security Rules Checklist
Dynamic Evaluation
Security Requirements:
  • ✖ At least 8 characters in length
  • ✖ At least 1 uppercase letter (A-Z)
  • ✖ At least 1 lowercase letter (a-z)
  • ✖ At least 1 numeric digit (0-9)
  • ✖ At least 1 special symbol (!@#$%^&*) 
function checkPasswordRequirements(password) {
  const rules = {
    length: password.length >= 8,
    upper: /[A-Z]/.test(password),
    lower: /[a-z]/.test(password),
    digit: /[0-9]/.test(password),
    special: /[!@#$%^&*(),.?":{}|<>]/.test(password)
  };

  // Update DOM classes based on boolean status
  updateCheckItem("ruleLength", rules.length);
  updateCheckItem("ruleUpper", rules.upper);
  updateCheckItem("ruleLower", rules.lower);
  updateCheckItem("ruleDigit", rules.digit);
  updateCheckItem("ruleSpecial", rules.special);

  const allPassed = Object.values(rules).every(Boolean);
  return allPassed;
}
