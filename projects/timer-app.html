let durationSeconds = 25 * 60;
let isRunning = false;
let pomoInterval = null;

function toggleTimer() {
  if (!isRunning) {
    isRunning = true;
    pomoInterval = setInterval(() => {
      if (durationSeconds <= 0) {
        clearInterval(pomoInterval);
        isRunning = false;
        alert("Session completed! Take a break.");
        return;
      }
      durationSeconds--;
      updatePomoDisplay();
    }, 1000);
  } else {
    clearInterval(pomoInterval);
    isRunning = false;
  }
}
