Interactive Body Mass Index (BMI) Health Calculator

A clinical biometric calculator supporting Metric and Imperial unit systems with interactive visual spectrum needles, WHO classification mapping, and ideal weight recommendations.

Biometric Formula Unit Switching (Metric/Imperial) Visual Spectrum Gauge Range Classifications

Educational Objective

Learn how to calculate Body Mass Index using the formulas: BMI = kg / (m)² (Metric) and BMI = 703 × lbs / (in)² (Imperial). Explore dynamic DOM attribute manipulation to move a visual gauge needle along a continuous spectrum bar and deliver personalized health diagnostics.

// BMI Computation Logic
let unit = 'metric';

function calculateBMI() {
  const weight = parseFloat(document.getElementById('inpWeight').value);
  const height = parseFloat(document.getElementById('inpHeight').value);

  if (isNaN(weight) || isNaN(height) || weight <= 0 || height <= 0) {
    alert('Please enter valid positive values for height and weight.');
    return;
  }

  let bmi = 0;
  if (unit === 'metric') {
    const heightM = height / 100;
    bmi = weight / (heightM * heightM);
  } else {
    bmi = (703 * weight) / (height * height);
  }

  const rounded = bmi.toFixed(1);
  document.getElementById('resBMI').textContent = rounded;

  let category = '';
  let color = '';
  let pct = 0;

  if (bmi < 18.5) {
    category = 'Underweight';
    color = '#38bdf8';
    pct = (bmi / 18.5) * 25;
  } else if (bmi < 25.0) {
    category = 'Normal Weight';
    color = '#10b981';
    pct = 25 + ((bmi - 18.5) / (24.9 - 18.5)) * 25;
  } else if (bmi < 30.0) {
    category = 'Overweight';
    color = '#f59e0b';
    pct = 50 + ((bmi - 25.0) / (29.9 - 25.0)) * 25;
  } else {
    category = 'Obese';
    color = '#ef4444';
    pct = Math.min(75 + ((bmi - 30.0) / 10.0) * 25, 100);
  }

  const resCat = document.getElementById('resCategory');
  resCat.textContent = category;
  resCat.style.color = color;
  document.getElementById('bmiNeedle').style.left = `${pct}%`;
}
