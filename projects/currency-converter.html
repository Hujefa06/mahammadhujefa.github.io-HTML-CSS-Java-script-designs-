Forex Exchange & Currency Converter

A financial currency converter providing instantaneous cross-rate foreign exchange calculations across major international currencies with dynamic one-click currency swapping and benchmark rate matrices.

Exchange Rate Matrix Currency Swapping Number & Decimal Formatting Reactive Input Events

Educational Objective

Understand triangular currency arbitrage and exchange rate mapping using a base currency anchor (USD). Master converting any source currency A to destination currency B via Amount × (Rate[B] / Rate[A]), implementing reactive event listeners for immediate recalculations.

// Currency Conversion Logic (Rates normalized to 1 USD)
const exchangeRates = {
  USD: 1.0,
  EUR: 0.92,
  GBP: 0.79,
  INR: 83.20,
  JPY: 155.40,
  CAD: 1.36,
  AUD: 1.52
};

function convertCurrency() {
  const amount = parseFloat(document.getElementById('inpAmount').value) || 0;
  const from = document.getElementById('selFrom').value;
  const to = document.getElementById('selTo').value;

  // Convert From -> USD -> To
  const rateInUSD = 1 / exchangeRates[from];
  const converted = amount * rateInUSD * exchangeRates[to];
  const singleUnitRate = (1 * rateInUSD * exchangeRates[to]);

  document.getElementById('dispFromSummary').textContent = `${amount.toLocaleString()} ${from} =`;
  document.getElementById('dispToResult').textContent = `${converted.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })} ${to}`;
  document.getElementById('dispRateNote').textContent = `1 ${from} = ${singleUnitRate.toFixed(4)} ${to} • Mid-market rate`;
}

document.getElementById('btnSwap').addEventListener('click', () => {
  const selFrom = document.getElementById('selFrom');
  const selTo = document.getElementById('selTo');
  const temp = selFrom.value;
  selFrom.value = selTo.value;
  selTo.value = temp;
  convertCurrency();
});

document.getElementById('inpAmount').addEventListener('input', convertCurrency);
document.getElementById('selFrom').addEventListener('change', convertCurrency);
document.getElementById('selTo').addEventListener('change', convertCurrency);

convertCurrency();
