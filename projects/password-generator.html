Cryptographic Random Password Generator

A security utility that produces cryptographically strong, customizable random passwords with character set selections, entropy strength analysis, and 1-click clipboard copying.

Crypto API / Math.random String Manipulation Clipboard API Password Entropy Meter

Educational Objective

Understand how randomized character selection functions by aggregating character sets into a unified lookup pool. Learn to enforce inclusion constraints (at least one symbol, digit, or uppercase character) and evaluate the resulting password strength based on character variance and string length.

// Password Generation Logic
const chars = {
  upper: 'ABCDEFGHIJKLMNOPQRSTUVWXYZ',
  lower: 'abcdefghijklmnopqrstuvwxyz',
  numbers: '0123456789',
  symbols: '!@#$%^&*()_+~`|}{[]:;?><,./-='
};

function generatePassword() {
  const length = parseInt(document.getElementById('passLength').value);
  const useUpper = document.getElementById('chkUpper').checked;
  const useLower = document.getElementById('chkLower').checked;
  const useNumbers = document.getElementById('chkNumbers').checked;
  const useSymbols = document.getElementById('chkSymbols').checked;

  let pool = '';
  let guaranteed = [];

  if (useUpper) { pool += chars.upper; guaranteed.push(chars.upper[Math.floor(Math.random() * chars.upper.length)]); }
  if (useLower) { pool += chars.lower; guaranteed.push(chars.lower[Math.floor(Math.random() * chars.lower.length)]); }
  if (useNumbers) { pool += chars.numbers; guaranteed.push(chars.numbers[Math.floor(Math.random() * chars.numbers.length)]); }
  if (useSymbols) { pool += chars.symbols; guaranteed.push(chars.symbols[Math.floor(Math.random() * chars.symbols.length)]); }

  if (!pool) {
    alert('Please select at least one character type!');
    return;
  }

  let result = [...guaranteed];
  for (let i = guaranteed.length; i < length; i++) {
    result.push(pool[Math.floor(Math.random() * pool.length)]);
  }

  // Shuffle array
  result.sort(() => Math.random() - 0.5);
  const password = result.join('');

  document.getElementById('passOutput').textContent = password;
  evaluateStrength(password, pool.length);
}

function evaluateStrength(pass, poolSize) {
  const bar = document.getElementById('strengthBar');
  const txt = document.getElementById('strengthText');

  let score = 0;
  if (pass.length >= 8) score++;
  if (pass.length >= 14) score++;
  if (poolSize > 36) score++;
  if (poolSize > 62) score++;

  if (score <= 1) {
    bar.style.width = '25%';
    bar.style.background = '#ef4444';
    txt.textContent = 'Weak';
    txt.style.color = '#ef4444';
  } else if (score === 2) {
    bar.style.width = '50%';
    bar.style.background = '#f59e0b';
    txt.textContent = 'Medium';
    txt.style.color = '#f59e0b';
  } else if (score === 3) {
    bar.style.width = '75%';
    bar.style.background = '#3b82f6';
    txt.textContent = 'Strong';
    txt.style.color = '#3b82f6';
  } else {
    bar.style.width = '100%';
    bar.style.background = '#10b981';
    txt.textContent = 'Very Strong';
    txt.style.color = '#10b981';
  }
}
