Secret Number Guessing Game (1 - 100)

Classic binary search style guessing challenge: the computer generates a pseudo-random integer between 1 and 100. Analyze user guesses with 'Too High' or 'Too Low' clues within 7 total tries.

let secretNumber = Math.floor(Math.random() * 100) + 1;
let remainingTries = 7;

function checkGuess(userGuess) {
  if (userGuess === secretNumber) {
    return { status: "win", message: "🎉 Correct! You won the game!" };
  }

  remainingTries--;
  if (remainingTries === 0) {
    return { status: "gameover", message: `Game Over! The number was ${secretNumber}` };
  }

  const hint = userGuess < secretNumber ? "📈 Too Low! Guess higher." : "📉 Too High! Guess lower.";
  return { status: "continue", message: hint, remainingTries };
}
