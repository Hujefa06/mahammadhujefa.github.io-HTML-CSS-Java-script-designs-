// Grade and SGPA Calculation
function calculateSGPA() {
  const rows = document.querySelectorAll('#courseBody tr');
  let totalCredits = 0;
  let totalQualityPoints = 0;

  rows.forEach(row => {
    const credit = parseFloat(row.querySelector('.c-credit').value) || 0;
    const grade = parseFloat(row.querySelector('.c-grade').value) || 0;
    totalCredits += credit;
    totalQualityPoints += (credit * grade);
  });

  const sgpa = totalCredits > 0 ? (totalQualityPoints / totalCredits) : 0;
  
  document.getElementById('dispSGPA').textContent = sgpa.toFixed(2);
  document.getElementById('dispCredits').textContent = totalCredits;
  document.getElementById('dispPoints').textContent = totalQualityPoints.toFixed(1);

  let honors = 'Pass';
  if (sgpa >= 9.0) honors = 'First Class with Distinction (Honors)';
  else if (sgpa >= 7.5) honors = 'First Class';
  else if (sgpa >= 6.0) honors = 'Second Class';
  else if (sgpa < 5.0 && totalCredits > 0) honors = 'Reappear Required';

  document.getElementById('dispClassification').textContent = honors;
}

function addCourseRow() {
  const tbody = document.getElementById('courseBody');
  const tr = document.createElement('tr');
  tr.innerHTML = `
    <td><input type="text" class="c-name" placeholder="Course Name"></td>
    <td>
      <select class="c-credit">
        <option value="4">4 Credits</option>
        <option value="3" selected>3 Credits</option>
        <option value="2">2 Credits</option>
        <option value="1">1 Credit</option>
      </select>
    </td>
    <td>
      <select class="c-grade">
        <option value="10">O (Outstanding - 10)</option>
        <option value="9">A+ (Excellent - 9)</option>
        <option value="8" selected>A (Very Good - 8)</option>
        <option value="7">B+ (Good - 7)</option>
        <option value="6">B (Above Average - 6)</option>
        <option value="5">C (Pass - 5)</option>
        <option value="0">F (Fail - 0)</option>
      </select>
    </td>
    <td><button class="btn-delete-row" onclick="removeCourseRow(this)">&times;</button></td>
  `;
  tbody.appendChild(tr);
  calculateSGPA();
}

function removeCourseRow(btn) {
  const tr = btn.closest('tr');
  if (document.querySelectorAll('#courseBody tr').length > 1) {
    tr.remove();
    calculateSGPA();
  } else {
    alert('At least one course is required.');
  }
}
