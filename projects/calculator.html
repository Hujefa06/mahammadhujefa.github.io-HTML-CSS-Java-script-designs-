Interactive Digital Calculator

A full-featured digital calculator supporting chained arithmetic calculations (+, -, *, /), percentage calculations, decimal handling, backspace, and physical keyboard event integration.

let currentInput = "0";
let previousInput = "";
let operation = null;

function appendNum(num) {
  if (currentInput === "0") currentInput = num;
  else currentInput += num;
  updateScreen();
}

function appendOperator(op) {
  if (operation !== null) calculateResult();
  previousInput = currentInput;
  operation = op;
  currentInput = "0";
}

function calculateResult() {
  const prev = parseFloat(previousInput);
  const curr = parseFloat(currentInput);
  let result = 0;

  switch (operation) {
    case '+': result = prev + curr; break;
    case '-': result = prev - curr; break;
    case '*': result = prev * curr; break;
    case '/': result = curr === 0 ? "Error" : prev / curr; break;
    case '%': result = prev % curr; break;
  }

  currentInput = String(result);
  operation = null;
  updateScreen();
}
