const cityDatabase = {
  "hyderabad": { temp: 29, cond: "Partly Cloudy", icon: "⛅", hum: 62, wind: 14, uv: "Moderate (5)" },
  "london": { temp: 16, cond: "Light Rain", icon: "🌧️", hum: 85, wind: 22, uv: "Low (2)" },
  "tokyo": { temp: 22, cond: "Clear Sky", icon: "☀️", hum: 45, wind: 10, uv: "High (7)" },
  "new york": { temp: 20, cond: "Breezy", icon: "💨", hum: 52, wind: 18, uv: "Moderate (4)" }
};

function fetchWeather(city) {
  const query = city.toLowerCase().trim();
  const data = cityDatabase[query];

  if (data) {
    displayWeather(data);
  } else {
    // Generate realistic meteorological simulation for any queried city
    const simTemp = Math.floor(Math.random() * 20 + 15);
    displayWeather({ temp: simTemp, cond: "Sunny Intervals", icon: "🌤️", hum: 55, wind: 12, uv: "5" });
  }
}
