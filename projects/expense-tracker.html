Personal Expense & Budget Tracker

A financial ledger web app that calculates real-time total balance, monthly income, and expenditures with categorized item records and persistent localStorage state.

Vanilla JS Array Methods DOM Dynamic Rendering localStorage JSON Math & Currency Formatting

Educational Objective

Understand how to model financial transactions using an array of structured objects { id, text, amount, category, type }. Master array calculation methods like .reduce() and .filter() to compute aggregated metrics on every mutation, while keeping data synced with the browser's localStorage.

// Expense Tracker Logic
const txForm = document.getElementById('txForm');
const txText = document.getElementById('txText');
const txAmount = document.getElementById('txAmount');
const txType = document.getElementById('txType');
const txList = document.getElementById('txList');
const netBalance = document.getElementById('netBalance');
const totalIncome = document.getElementById('totalIncome');
const totalExpense = document.getElementById('totalExpense');

let transactions = JSON.parse(localStorage.getItem('weblab_expenses')) || [
  { id: 1, text: 'Freelance Project', amount: 850.00, type: 'income' },
  { id: 2, text: 'Grocery Store', amount: 64.50, type: 'expense' },
  { id: 3, text: 'Coffee & Snacks', amount: 12.00, type: 'expense' }
];

function updateUI() {
  txList.innerHTML = '';
  
  if (transactions.length === 0) {
    txList.innerHTML = '<li class="empty-state">No transactions logged yet.</li>';
  } else {
    transactions.forEach(tx => {
      const li = document.createElement('li');
      li.className = `history-item ${tx.type}`;
      li.innerHTML = `
        <div class="item-info">
          <span class="item-desc">${escapeHTML(tx.text)}</span>
          <span class="item-category">${tx.type}</span>
        </div>
        <div class="item-right">
          <span class="item-amount">${tx.type === 'income' ? '+' : '-'}$${tx.amount.toFixed(2)}</span>
          <button class="item-delete" onclick="deleteTx(${tx.id})" title="Delete">×</button>
        </div>
      `;
      txList.appendChild(li);
    });
  }

  const income = transactions
    .filter(t => t.type === 'income')
    .reduce((acc, t) => acc + t.amount, 0);

  const expense = transactions
    .filter(t => t.type === 'expense')
    .reduce((acc, t) => acc + t.amount, 0);

  const balance = income - expense;

  netBalance.textContent = (balance >= 0 ? '$' : '-$') + Math.abs(balance).toFixed(2);
  totalIncome.textContent = `+$${income.toFixed(2)}`;
  totalExpense.textContent = `-$${expense.toFixed(2)}`;

  localStorage.setItem('weblab_expenses', JSON.stringify(transactions));
}

txForm.addEventListener('submit', (e) => {
  e.preventDefault();
  const text = txText.value.trim();
  const amount = parseFloat(txAmount.value);
  if (!text || isNaN(amount) || amount <= 0) return;

  transactions.unshift({
    id: Date.now(),
    text,
    amount,
    type: txType.value
  });

  txText.value = '';
  txAmount.value = '';
  updateUI();
});

window.deleteTx = function(id) {
  transactions = transactions.filter(t => t.id !== id);
  updateUI();
};

document.getElementById('clearAllBtn').addEventListener('click', () => {
  if (confirm('Are you sure you want to clear all transactions?')) {
    transactions = [];
    updateUI();
  }
});

function escapeHTML(str) {
  return str.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}

updateUI();
