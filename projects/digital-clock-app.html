function getTimeInZone(timeZone) {
  const now = new Date();
  
  // Format localized time string
  const timeStr = now.toLocaleTimeString("en-US", {
    timeZone,
    hour12: true,
    hour: "2-digit",
    minute: "2-digit",
    second: "2-digit"
  });

  // Format localized date string
  const dateStr = now.toLocaleDateString("en-US", {
    timeZone,
    weekday: "short",
    month: "short",
    day: "numeric"
  });

  return { timeStr, dateStr };
}
