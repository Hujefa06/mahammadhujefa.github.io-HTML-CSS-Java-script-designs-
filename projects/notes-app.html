// Sticky Notes Logic
let notes = JSON.parse(localStorage.getItem('weblab_notes')) || [
  { id: 1, title: 'Web Lab Project', content: 'Complete all 150 experiments with interactive code viewers!', color: 'yellow', date: 'Sep 21, 2026' },
  { id: 2, title: 'Exam Revision', content: 'Review DOM tree traversal, event bubbling, and closures.', color: 'blue', date: 'Sep 21, 2026' }
];

let activeColor = 'yellow';

function renderNotes(filterText = '') {
  const grid = document.getElementById('notesGrid');
  const filtered = notes.filter(n => 
    n.title.toLowerCase().includes(filterText.toLowerCase()) || 
    n.content.toLowerCase().includes(filterText.toLowerCase())
  );

  if (filtered.length === 0) {
    grid.innerHTML = '<div class="empty-notes-hint">No notes found. Create a new one!</div>;
    return;
  }

  grid.innerHTML = filtered.map(n => `
    <div class="note-card ${n.color}" data-id="${n.id}">
      <div class="note-title" contenteditable="true" onblur="updateNote(${n.id}, 'title', this.innerText)">${escapeHTML(n.title)}</div>
      <div class="note-body" contenteditable="true" onblur="updateNote(${n.id}, 'content', this.innerText)">${escapeHTML(n.content)}</div>
      <div class="note-footer">
        <span>${n.date}</span>
        <button class="note-del-btn" onclick="deleteNote(${n.id})" title="Delete">&times;</button>
      </div>
    </div>
  `).join('');
}

function updateNote(id, key, val) {
  const note = notes.find(n => n.id === id);
  if (note) {
    note[key] = val.trim();
    saveNotes();
  }
}

function createNote() {
  const newNote = {
    id: Date.now(),
    title: 'Note Title',
    content: 'Type your note text here...',
    color: activeColor,
    date: new Date().toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
  };
  notes.unshift(newNote);
  saveNotes();
  renderNotes();
}

function deleteNote(id) {
  notes = notes.filter(n => n.id !== id);
  saveNotes();
  renderNotes();
}

function saveNotes() {
  localStorage.setItem('weblab_notes', JSON.stringify(notes));
}
