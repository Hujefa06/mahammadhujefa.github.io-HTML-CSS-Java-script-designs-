Live Word & Character Counter with Limit Alert

Calculate character count, word density, sentences, and estimated reading time in real-time as the user types, highlighting warnings when exceeding length limits.

const textarea = document.getElementById("liveTextArea");
const MAX_LIMIT = 250;

function updateMetrics() {
  const text = textarea.value;

  // 1. Character count
  const chars = text.length;

  // 2. Word count (filter out empty strings)
  const words = text.trim() === "" ? 0 : text.trim().split(/\s+/).length;

  // 3. Sentence count (split by punctuation: . ! ?)
  const sentences = text.trim() === "" ? 0 : (text.match(/[\w|\)][.?!]+(\s|$)/g) || []).length;

  // 4. Reading time estimate (200 words per minute average)
  const readSeconds = Math.ceil((words / 200) * 60);

  // 5. Exceeded limit check
  if (chars > MAX_LIMIT) {
    textarea.classList.add("limit-warning");
  } else {
    textarea.classList.remove("limit-warning");
  }
}
