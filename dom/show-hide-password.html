Show / Hide Password Input Toggle

Toggle password visibility dynamically by switching the input type attribute between "password" and "text" with icon feedback and strength estimation.

const pwdInput = document.getElementById("pwdInput");
const toggleBtn = document.getElementById("pwdToggleBtn");

function togglePasswordVisibility() {
  // Check current input type
  if (pwdInput.type === "password") {
    pwdInput.type = "text";
    toggleBtn.textContent = "🙈"; // Hide icon
  } else {
    pwdInput.type = "password";
    toggleBtn.textContent = "👁️"; // Show icon
  }
}
