classList API: add(), remove(), toggle(), contains()

Modern class manipulation in JavaScript without fragile string concatenations using the HTML5 DOMTokenList (classList) interface.

targetBox.className: sample-box 
const box = document.getElementById("targetBox");

// 1. add() - appends a class if not present
box.classList.add("theme-highlight");

// 2. remove() - removes the specified class
box.classList.remove("theme-highlight");

// 3. toggle() - adds if missing, removes if present
function toggleHighlight() {
  box.classList.toggle("theme-highlight");
  console.log("Active classes:", box.classList.value);
}

// 4. contains() - returns boolean true/false
function checkHighlight() {
  const isPresent = box.classList.contains("theme-highlight");
  alert("Is 'theme-highlight' applied? " + isPresent);
}

// 5. replace() - swaps one class for another
box.classList.replace("old-class", "new-class");
