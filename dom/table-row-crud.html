Student Records Table CRUD (Create, Read, Update, Delete)

Perform complete in-memory CRUD operations directly on DOM table rows: adding new entries, inline editing cell data, updating records, and removing table rows.

let students = [
  { roll: "22CSE01", name: "Shaik Roshan", dept: "CSE", marks: 95 },
  { roll: "22CSE02", name: "Ananya Patel", dept: "IT", marks: 88 }
];

function renderTable() {
  const tbody = document.getElementById("tableBody");
  tbody.innerHTML = "";

  students.forEach((s, i) => {
    const tr = tbody.insertRow();
    tr.innerHTML = `
      <td>${i + 1}</td>
      <td>${s.roll}</td>
      <td>${s.name}</td>
      <td>${s.dept}</td>
      <td>${s.marks}%</td>
      <td>
        <button class="btn-edit" onclick="editRow(${i})">Edit</button>
        <button class="btn-del" onclick="deleteRow(${i})">Delete</button>
      </td>
    `;
  });
}

function deleteRow(index) {
  students.splice(index, 1);
  renderTable();
}
