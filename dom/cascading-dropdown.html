Dependent Cascading Dropdowns (Country → State → City)

Dynamically populate dependent dropdown lists using structured nested JavaScript objects. Selecting a country loads its states and clears child city dropdowns automatically.

const locationData = {
  "India": {
    "Telangana": ["Hyderabad", "Warangal", "Nizamabad"],
    "Karnataka": ["Bengaluru", "Mysuru", "Hubballi"],
    "Maharashtra": ["Mumbai", "Pune", "Nagpur"]
  },
  "USA": {
    "California": ["Los Angeles", "San Francisco", "San Diego"],
    "Texas": ["Austin", "Houston", "Dallas"]
  }
};

function onCountrySelected() {
  const country = countrySelect.value;
  stateSelect.innerHTML = '';
  citySelect.innerHTML = '';
  citySelect.disabled = true;

  if (country && locationData[country]) {
    stateSelect.disabled = false;
    Object.keys(locationData[country]).forEach(state => {
      stateSelect.add(new Option(state, state));
    });
  } else {
    stateSelect.disabled = true;
  }
}
