const slides = [
  { img: "mountain.jpg", title: "Alpine Ridge", desc: "Glacier view" },
  { img: "sea.jpg", title: "Coastal Sunset", desc: "Pacific waters" },
  { img: "forest.jpg", title: "Pine Forest", desc: "Morning mist" }
];

let currentIndex = 0;
const imgEl = document.getElementById("slideImg");
const titleEl = document.getElementById("slideTitle");

function showSlide(index) {
  if (index < 0) index = slides.length - 1;
  if (index >= slides.length) index = 0;
  currentIndex = index;

  imgEl.src = slides[currentIndex].img;
  titleEl.textContent = slides[currentIndex].title;
}

function nextSlide() { showSlide(currentIndex + 1); }
function prevSlide() { showSlide(currentIndex - 1); }

// Auto-play interval
let timer = setInterval(nextSlide, 3500);
