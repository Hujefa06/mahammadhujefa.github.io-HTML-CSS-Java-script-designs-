const box = document.getElementById("paletteBox");
const hexDisplay = document.getElementById("hexDisplay");

// Apply color and update text
function setColor(hex) {
  box.style.backgroundColor = hex;
  hexDisplay.textContent = hex.toUpperCase();
}

// Generate random 6-character hex code
function generateRandomColor() {
  const letters = "0123456789ABCDEF";
  let color = "#";
  for (let i = 0; i < 6; i++) {
    color += letters[Math.floor(Math.random() * 16)];
  }
  setColor(color);
}

// Copy color hex to clipboard
function copyHexCode() {
  navigator.clipboard.writeText(hexDisplay.textContent);
  alert("Copied " + hexDisplay.textContent + " to clipboard!");
}
