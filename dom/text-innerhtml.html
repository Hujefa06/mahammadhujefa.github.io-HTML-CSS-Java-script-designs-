<script>
const el = document.getElementById("sampleNode");

// 1. textContent retrieves all text regardless of CSS visibility
console.log(el.textContent);

// 2. innerText respects CSS styles (excludes display: none)
console.log(el.innerText);

// 3. innerHTML parses and renders tags, or returns raw markup
console.log(el.innerHTML);
