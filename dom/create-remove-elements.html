const container = document.getElementById("nodesContainer");

function createCard(text) {
  // 1. Create wrapper element
  const node = document.createElement("div");
  node.className = "node-item";

  // 2. Create label span
  const span = document.createElement("span");
  span.textContent = text;
  node.appendChild(span);

  // 3. Create delete button
  const delBtn = document.createElement("button");
  delBtn.className = "del-btn";
  delBtn.innerHTML = "×";
  delBtn.onclick = function() {
    node.remove(); // removes node from DOM tree
  };
  node.appendChild(delBtn);

  return node;
}

// Append child at the end
function appendNode(text) {
  const node = createCard(text);
  container.appendChild(node);
}

// Prepend child at the beginning
function prependNode(text) {
  const node = createCard(text);
  container.prepend(node);
}
