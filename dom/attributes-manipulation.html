const el = document.getElementById("targetElement");

// 1. Get an attribute value
const title = el.getAttribute("title");

// 2. Set an attribute
el.setAttribute("title", "Updated Premium Audio Product");
el.setAttribute("data-price", "$249.99");

// 3. Check if attribute exists
if (el.hasAttribute("data-category")) {
  console.log("Category exists:", el.getAttribute("data-category"));
}

// 4. Remove attribute
el.removeAttribute("data-category");

// 5. HTML5 dataset property access
console.log(el.dataset.itemId); // Accesses 'data-item-id'
