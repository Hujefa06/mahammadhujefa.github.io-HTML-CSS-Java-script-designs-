The DOM is an object-oriented programmatic representation of the web page. It acts as an API for JavaScript to navigate, inspect, add, alter, and delete HTML elements, attributes, and CSS styling dynamically at runtime.
In capturing phase, the event descends from the outermost window/document down to the target element. In bubbling phase (the browser default), the event bubbles up from the innermost clicked element back up to window.
addEventListener() allows registering multiple distinct event handlers for the exact same event without overwriting previous handlers, supports event capturing phases, and facilitates clean detachment via removeEventListener().
localStorage persists data indefinitely across browser sessions until explicitly cleared. sessionStorage only retains data during the active browser tab lifetime and gets destroyed upon closing the tab. 
function toggleAccordion(btn) {
  const item = btn.parentElement;
  const singleMode = document.getElementById("singleModeCheckbox").checked;

  if (singleMode) {
    // Close all other sibling items
    document.querySelectorAll(".accordion-item").forEach(otherItem => {
      if (otherItem !== item) {
        otherItem.classList.remove("open");
      }
    });
  }

  // Toggle current item
  item.classList.toggle("open");
}
