Dynamic HTML Table Generation from Inputs

Generate an N-row by M-column structured HTML table using JavaScript DOM loops, demonstrating table.insertRow() and row.insertCell() methods.

function generateTable(rows, cols) {
  const container = document.getElementById("tableContainer");
  container.innerHTML = ""; // clear previous

  const table = document.createElement("table");
  table.className = "generated-table";

  // Create Header Row
  const thead = table.createTHead();
  const headerRow = thead.insertRow();
  for (let c = 1; c <= cols; c++) {
    const th = document.createElement("th");
    th.textContent = `Col ${c}`;
    headerRow.appendChild(th);
  }

  // Create Table Body
  const tbody = table.createTBody();
  for (let r = 1; r <= rows; r++) {
    const row = tbody.insertRow();
    for (let c = 1; c <= cols; c++) {
      const cell = row.insertCell();
      cell.textContent = `[${r}, ${c}]`;
    }
  }

  container.appendChild(table);
}
