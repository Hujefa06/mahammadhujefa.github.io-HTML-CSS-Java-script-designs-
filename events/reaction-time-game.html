let state = "idle"; // "idle" | "waiting" | "ready" | "result"
let startTime = 0;
let timeoutId = null;

function handleBoxClick() {
  if (state === "idle" || state === "result") {
    // Start test: turn red and wait random delay (2 to 5 seconds)
    state = "waiting";
    box.className = "reaction-target-box state-waiting";
    box.innerText = "Wait for Green...";

    const delay = Math.random() * 3000 + 2000;
    timeoutId = setTimeout(() => {
      state = "ready";
      startTime = performance.now();
      box.className = "reaction-target-box state-ready";
      box.innerText = "CLICK NOW!";
    }, delay);

  } else if (state === "waiting") {
    // Clicked too early!
    clearTimeout(timeoutId);
    state = "result";
    box.innerText = "Too early! Click to try again.";
  } else if (state === "ready") {
    // Successful reaction click
    const reactionTime = Math.round(performance.now() - startTime);
    state = "result";
    box.innerText = `${reactionTime} ms! Click to try again.`;
  }
}
