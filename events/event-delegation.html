Event Delegation Pattern using e.target.closest()

Bind a single event listener to a common parent element instead of attaching hundreds of separate listeners to dynamic children. Works seamlessly for newly created DOM nodes without re-binding.

Single Listener on <ul>
Notice: Parent <ul id="delegatedList"> has only ONE click listener handling all buttons!

const parentList = document.getElementById("delegatedList");

// Single event listener on the PARENT container
parentList.addEventListener("click", (event) => {
  const target = event.target;
  const action = target.getAttribute("data-action");
  const row = target.closest(".list-item-btn");

  if (!action || !row) return;

  const itemId = row.getAttribute("data-id");

  if (action === "delete") {
    row.remove(); // Removes the clicked row
    console.log(`Deleted item ${itemId}`);
  } else if (action === "view") {
    alert(`Viewing details for Item #${itemId}`);
  }
});
