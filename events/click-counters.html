Click Speed Test (CPS Meter with Countdown Timer)

Calculate rapid clicking frequency (Clicks Per Second) using high precision timestamps and interval countdown timers, demonstrating stateful event handling.

let clicks = 0;
let timeLeft = 5;
let timer = null;
let isRunning = false;

function handleClick() {
  if (!isRunning && timeLeft === 5) {
    isRunning = true;
    startTimer();
  }

  if (isRunning) {
    clicks++;
    clicksEl.textContent = clicks;
  }
}

function startTimer() {
  timer = setInterval(() => {
    timeLeft -= 0.1;
    timerEl.textContent = timeLeft.toFixed(1) + "s";

    if (timeLeft <= 0) {
      clearInterval(timer);
      isRunning = false;
      const cps = (clicks / 5).toFixed(2);
      alert(`Test Complete! Your CPS is: ${cps}`);
    }
  }, 100);
}
