const radar = document.getElementById("radarBox");
const spotlight = document.getElementById("spotlight");

radar.addEventListener("mousemove", (event) => {
  // 1. Element relative
  const ox = event.offsetX;
  const oy = event.offsetY;

  // 2. Viewport relative
  const cx = event.clientX;
  const cy = event.clientY;

  // 3. Document page relative (includes scroll)
  const px = event.pageX;
  const py = event.pageY;

  // Position spotlight
  spotlight.style.left = `${ox}px`;
  spotlight.style.top = `${oy}px`;
});
