function calculateMetrics(typedChars, totalErrors, timeElapsedMinutes) {
  // Gross Words = total characters / 5
  const grossWPM = (typedChars / 5) / timeElapsedMinutes;
  
  // Net WPM = (total characters / 5 - errors) / timeElapsedMinutes
  const netWPM = Math.max(0, Math.round(((typedChars / 5) - totalErrors) / timeElapsedMinutes));

  // Accuracy Percentage
  const accuracy = Math.round(((typedChars - totalErrors) / typedChars) * 100);

  return { netWPM, accuracy };
}
