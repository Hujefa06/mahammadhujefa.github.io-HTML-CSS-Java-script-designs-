window.addEventListener("keydown", (event) => {
  // 1. Modern string representation of key value
  const key = event.key;

  // 2. Physical key on the keyboard hardware
  const code = event.code;

  // 3. Numeric keyCode (deprecated in modern web, but useful in legacy code)
  const keyCode = event.which || event.keyCode;

  // 4. Modifier keys
  const isCtrl = event.ctrlKey;
  const isShift = event.shiftKey;
  const isAlt = event.altKey;

  console.log({ key, code, keyCode, isCtrl, isShift, isAlt });
});
