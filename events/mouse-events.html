const pad = document.getElementById("mousePad");

// 1. Single Click
pad.addEventListener("click", (e) => logEvent("click", `Button: ${e.button}`));

// 2. Double Click
pad.addEventListener("dblclick", () => logEvent("dblclick", "Double Click detected"));

// 3. Right Click (Context Menu)
pad.addEventListener("contextmenu", (e) => {
  e.preventDefault(); // Prevent standard browser context menu
  logEvent("contextmenu", "Custom Right Click handled");
});

// 4. Hover lifecycle
pad.addEventListener("mouseenter", () => logEvent("mouseenter", "Cursor entered target"));
pad.addEventListener("mouseleave", () => logEvent("mouseleave", "Cursor left target"));
pad.addEventListener("mousedown", () => logEvent("mousedown", "Mouse button pressed down"));
pad.addEventListener("mouseup", () => logEvent("mouseup", "Mouse button released"));
