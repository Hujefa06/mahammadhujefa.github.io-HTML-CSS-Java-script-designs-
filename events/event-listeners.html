[Ready] Handlers initialized on target element.
const btn = document.getElementById("targetBtn");

// Named handler functions (required for removeEventListener)
function flashHandler() {
  console.log("Flash handler fired");
}

function counterHandler() {
  console.log("Counter incremented");
}

// 1. Adding multiple listeners
btn.addEventListener("click", flashHandler);
btn.addEventListener("click", counterHandler);

// 2. Removing a specific listener
btn.removeEventListener("click", flashHandler);

// 3. One-time listener with option object
btn.addEventListener("click", () => {
  console.log("Fires exactly once, then automatically unbinds!");
}, { once: true });
