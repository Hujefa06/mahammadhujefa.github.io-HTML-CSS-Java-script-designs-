Keyboard Arrow & WASD 2D Character Controller

Capture continuous keyboard events (Arrow keys & WASD) to move a 2D game sprite inside an arena boundary with boundary clamping, score tracking, and coin collision detection.

⌨️ Controls: Use Arrow Keys or W, A, S, D on your keyboard, or click the on-screen D-Pad. 
let posX = 150, posY = 150;
const STEP = 20;
const avatar = document.getElementById("avatar");

window.addEventListener("keydown", (e) => {
  switch (e.key) {
    case "ArrowUp":
    case "w":
    case "W":
      posY -= STEP;
      e.preventDefault();
      break;
    case "ArrowDown":
    case "s":
    case "S":
      posY += STEP;
      e.preventDefault();
      break;
    case "ArrowLeft":
    case "a":
    case "A":
      posX -= STEP;
      e.preventDefault();
      break;
    case "ArrowRight":
    case "d":
    case "D":
      posX += STEP;
      e.preventDefault();
      break;
  }
  updatePosition();
});
